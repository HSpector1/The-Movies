function ownPromise(state: GameState, personId: string): ProfessionalPromise {
  const proposal = currentProposals(state, personId).find((p) => p.issuerStudioId === player(state))
  assert.ok(proposal)
  assert.equal(proposal.promises.length, 1)
  const promise = state.promises.find((p) => p.promiseId === proposal.promises[0])
  assert.ok(promise)
  return promise
}
function binding(state: GameState, promise: ProfessionalPromise): void {
  assert.notEqual(promise.contractId, null)
  const employment = state.hollywood!.employment.find((row) => row.contractId === promise.contractId)
  assert.ok(employment)
  assert.equal(employment.studioId, promise.issuerStudioId)
  assert.equal(employment.terms.talentId, promise.beneficiaryPersonId)
  assert.equal(employment.terms.startWeek, promise.windowStartWeek)
  assert.ok(state.talentMarket.receipts.some((r) => r.kind === 'settled' && r.week === employment.terms.startWeek
    && r.talentId === promise.beneficiaryPersonId && r.studioId === promise.issuerStudioId))
}
function outcome(state: GameState, promise: ProfessionalPromise): void {
  binding(state, promise)
  assert.notEqual(promise.outcome, null)
  assert.notEqual(promise.outcomeEventId, null)
  const refs = state.talentMarket.receipts.filter((r) => r.eventId === promise.outcomeEventId)
  assert.equal(refs.length, 1)
  assert.equal(refs[0]!.kind, 'promiseOutcome')
  assert.equal(refs[0]!.week, promise.outcomeWeek)
  assert.equal(refs[0]!.talentId, promise.beneficiaryPersonId)
  assert.equal(refs[0]!.studioId, promise.issuerStudioId)
}
function bridgeAction(state: GameState, draft: BridgeMarketProposalDraftPayload, id: string) {
  const session = new BridgeSession(structuredClone(state), `v29-before-p2-${id}`)
  const controls = (commandId: string) => ({ protocolVersion: PROTOCOL_VERSION, schemaId: SCHEMA_ID,
    sessionId: session.sessionId, commandId, expectedStateRevision: session.stateRevision })
  const quoted = session.quote({ ...controls(`quote-${id}`), type: 'quoteMarketProposal', draft })
  if (!quoted.accepted) throw new Error(quoted.message)
  assert.equal(quoted.quote.ok, true)
  const committed = session.command({ ...controls(`commit-${id}`), type: 'submitIntent', payload: { intentId: quoted.quote.intentId } })
  assert.ok(committed.accepted, committed.message)
  assert.equal(session.stateRevision, 1)
  assert.equal(session.gameState.studio.cash, state.studio.cash)
  assert.deepEqual(session.gameState.ledger, state.ledger)
  assert.equal(session.gameState.rngState, state.rngState)
  return { state: session.gameState, recipe: { source: 'actual BridgeSession quote and commit on accepted final-V29 producer', draft,
    intentId: quoted.quote.intentId, initialRevision: 0, finalRevision: 1 } }
}
