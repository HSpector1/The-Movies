    expect(record!.status, 'commitRetirementExtension alone must not retire or reclassify the person').toBe('announced')
  })

  it('W5b: end-to-end at the decision week (axis b, the literal "no gap, no overlap" case): the case settles, a new contract runs decisionWeek -> newE, and the signing bonus is charged exactly once', () => {
    const state = withExtensionCase(c2bFixture(AXIS_B.fixture), {

    const settled = advanceTo(withOffer, AXIS_B.decisionWeek)
