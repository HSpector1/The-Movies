export const sha = (value: string | Uint8Array): string => createHash('sha256').update(value).digest('hex')
export const gitRaw = (...args: string[]): string => execFileSync('git', args, { encoding: 'utf8' })
export const git = (...args: string[]): string => gitRaw(...args).trim()

export function gate(authority: MinterAuthority) {
  assert.match(authority.testedSourceSha, /^[a-f0-9]{40}$/, `${authority.phase} accepted source is not pinned`)
  assert.match(authority.textOnlySourceSha, /^[a-f0-9]{40}$/, `${authority.phase} text-only pass is not pinned`)
  assert.match(authority.textOnlyProducerDiffSha256, /^[a-f0-9]{64}$/, 'text-only producer diff is not pinned')
  assert.match(authority.publishedRecoverySha, /^[a-f0-9]{40}$/, `${authority.phase} accepted publication is not pinned`)
  assert.match(authority.closeoutSha256, /^[a-f0-9]{64}$/, 'accepted closeout evidence is not pinned')
  assert.equal(process.env[authority.approvalEnvironmentVariable], authority.publishedRecoverySha,
    `REFUSED: ${authority.approvalEnvironmentVariable} must equal the independently accepted publication `
    + `${authority.publishedRecoverySha}; only the parent sets it, and only for the one-shot mint`)
  const root = git('rev-parse', '--show-toplevel')
  assert.equal(resolve(process.cwd()), root, 'invoke only at the authorized repository root')
  assert.equal(git('branch', '--show-current'), BRANCH)
  assert.equal(git('rev-parse', 'HEAD'), authority.publishedRecoverySha, 'producer moved after acceptance')
  git('merge-base', '--is-ancestor', authority.testedSourceSha, authority.textOnlySourceSha)
  git('merge-base', '--is-ancestor', authority.textOnlySourceSha, authority.publishedRecoverySha)
  // No producer byte moved after the text pass; the text pass itself is exactly the reviewed comment-only diff.
  git('diff', '--exit-code', authority.textOnlySourceSha, '--', ...PRODUCER_PATHS)
  assert.equal(sha(gitRaw('diff', '--no-color', '--no-ext-diff', authority.testedSourceSha, authority.textOnlySourceSha, '--', ...PRODUCER_PATHS)),
    authority.textOnlyProducerDiffSha256, 'the behavioural-writer → text-pass producer diff is not the reviewed comment-only diff')
  assert.equal(git('ls-files', '--others', '--exclude-standard', '--', ...PRODUCER_PATHS), '', 'untracked producer code is forbidden')
  assert.equal(LIVE_SAVE_VERSION, authority.saveVersion, 'not the required outgoing V30 writer')
  assert.equal(PROTOCOL_VERSION, authority.protocolVersion)
  assert.equal(PROJECTION_VERSION, authority.projectionVersion)
  assert.equal(SCHEMA_ID, authority.schemaId)
  assert.equal(PROMISE_RULES_VERSION, authority.promiseRulesVersion, 'wrong promise evaluator generation')
  for (const path of [authority.closeoutRelativePath, authority.outputRelativePath]) {
    assert.ok(!path.includes('..') && !path.startsWith('/'), 'paths must stay relative inside the authorized repository')
  }
  assert.ok(authority.outputRelativePath.startsWith('tests/fixtures/p14/'), 'output must remain the scoped new fixture corpus')
  const closeoutHash = sha(readFileSync(join(root, authority.closeoutRelativePath)))
  assert.equal(closeoutHash, authority.closeoutSha256)
  assert.ok(!existsSync(join(root, authority.outputRelativePath)), 'refusing existing/partial corpus directory')
  const sourceFiles = Object.fromEntries(EVIDENCE_FILES.map((path) => [path, sha(readFileSync(join(root, path)))]))
  return { root, sourceFiles, closeoutHash }
}

export type Prepared = ReturnType<typeof prepareV30Corpus>[number]
/** Every world, readback and hash, in memory. Writes nothing; needs no token. */
export function prepareV30Corpus(count: number, build: () => Snapshot[]) {
  const prepared = build().map((item) => {
    assert.match(item.name, /^[a-z0-9-]+$/)
    const raw = exportSave(makeSave(item.state))
    const validated = validateSaveV30(JSON.parse(raw))
    assert.equal(validated.saveVersion, 30)
    assert.equal(exportSave(validated), raw)
    assert.equal(exportSave(validateSaveV30(importSave(raw))), raw)
    const session = BridgeSession.fromSaveJson(raw, `v30-roundtrip-${item.name}`)
    assert.equal(exportSave(makeSave(session.gameState)), raw)
    assert.deepEqual(session.gameState.promises, item.state.promises)
    assert.deepEqual(session.gameState.firstTakes, item.state.firstTakes)
    assert.deepEqual(session.gameState.talentMarket, item.state.talentMarket)
    const compressed = gzipSync(raw, { level: 9 })
    assert.equal(gunzipSync(compressed).toString('utf8'), raw)
    return { item, compressed, facts: { seed: item.state.seed, week: item.state.market.tick, saveVersion: 30,
      uncompressedSha256: sha(raw), compressedSha256: sha(compressed), byteLength: Buffer.byteLength(raw),
      compressedByteLength: compressed.length,
      rootCounts: { promises: item.state.promises.length, firstTakes: item.state.firstTakes.length,
        marketReceipts: item.state.talentMarket.receipts.length, currentProposals: item.state.talentMarket.proposals.length },
      focus: item.state.promises.filter((p) => item.focusPromiseIds.includes(p.promiseId)).map((p) => ({
        promiseId: p.promiseId, family: p.family, predicate: p.predicate, version: p.version,
        issuerStudioId: p.issuerStudioId, beneficiaryPersonId: p.beneficiaryPersonId,
        windowStartWeek: p.windowStartWeek, dueWeekExclusive: p.dueWeekExclusive,
        contractId: p.contractId, outcome: p.outcome, outcomeWeek: p.outcomeWeek,
        outcomeEventId: p.outcomeEventId, evidenceRefs: p.evidenceRefs, originalFeasibilityReceipt: p.feasibilityReceipt })) } }
  })
  assert.equal(prepared.length, count)
  assert.equal(new Set(prepared.map((p) => p.item.name)).size, count)
  for (const { item, facts } of prepared) assert.equal(facts.focus.length, item.focusPromiseIds.length, `${item.name}: a focus id is not a root`)
  return prepared
}

export function mintV30Corpus(authority: MinterAuthority, minterUrl: string, count: number, build: () => Snapshot[]): void {
  const initial = gate(authority)
  const minterPath = fileURLToPath(minterUrl)
  assert.ok(minterPath.startsWith(`${initial.root}/tests/`), 'install operational entry only in authorized tests path')
  const minterRelativePath = minterPath.slice(initial.root.length + 1)
  const operationalHashes = { minterSha256: sha(readFileSync(new URL(minterUrl))),
    supportSha256: sha(readFileSync(new URL(import.meta.url))) }
  const startedAt = new Date().toISOString()
  // Prepare ALL worlds and readbacks before any output creation. No JSON restamp.
  const prepared = prepareV30Corpus(count, build)
  const final = gate(authority)
  assert.deepEqual(final.sourceFiles, initial.sourceFiles, 'producer/recipe/configuration bytes moved during generation')
  assert.equal(final.closeoutHash, initial.closeoutHash)
  assert.equal(sha(readFileSync(new URL(minterUrl))), operationalHashes.minterSha256, 'executed entry changed during mint')
  assert.equal(sha(readFileSync(new URL(import.meta.url))), operationalHashes.supportSha256, 'shared helper changed during mint')
  const common = { authority, observedHeadSha: git('rev-parse', 'HEAD'), sourceFiles: initial.sourceFiles,
    ...operationalHashes,
    nodeVersion: process.version, startedAt, endedAt: new Date().toISOString(),
    campaignSource: 'generated test campaigns only; never Owner saves',
    command: `${authority.approvalEnvironmentVariable}=${authority.publishedRecoverySha} node_modules/.bin/vitest run ${minterRelativePath} --minWorkers=1 --maxWorkers=1`,
    scope: 'genuine SaveV30 producer evidence; no forward migration or native/product acceptance claim' }
  const directory = join(initial.root, authority.outputRelativePath)
  mkdirSync(directory) // fresh exact directory; no recursive delete/overwrite
  const manifest = []
  for (const { item, compressed, facts } of prepared) {
    const filename = `genuine-v30-${item.name}.json.gz`
    const provenanceFilename = `genuine-v30-${item.name}.provenance.json`
    writeFileSync(join(directory, filename), compressed, { flag: 'wx' })
    writeFileSync(join(directory, provenanceFilename), JSON.stringify({ ...common, filename,
      role: item.role, recipe: item.recipe, ...facts, additionalFacts: item.facts ?? null }, null, 2) + '\n', { flag: 'wx' })
    manifest.push({ filename, provenanceFilename, ...facts })
  }
  // LAST: absence means an incomplete batch, not permission to overwrite/remint.
  writeFileSync(join(directory, 'MANIFEST.json'), JSON.stringify({ ...common, fixtures: manifest }, null, 2) + '\n', { flag: 'wx' })
  console.log(JSON.stringify({ corpus: authority.outputRelativePath, observedHeadSha: common.observedHeadSha,
    fixtures: manifest.map(({ filename, uncompressedSha256, compressedSha256 }) => ({ filename, uncompressedSha256, compressedSha256 })) }, null, 2))
}
