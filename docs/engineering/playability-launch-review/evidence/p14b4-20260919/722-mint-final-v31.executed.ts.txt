function root(state: GameState, promiseId: string): ProfessionalPromise {
  const rows = state.promises.filter((p) => p.promiseId === promiseId)
  assert.equal(rows.length, 1)
  return rows[0]!
}
function ownPromise(state: GameState, personId: string): ProfessionalPromise {
  const proposal = currentProposals(state, personId).find((p) => p.issuerStudioId === player(state))
  assert.ok(proposal)
  assert.equal(proposal.promises.length, 1)
  return root(state, proposal.promises[0]!)
}
function binding(state: GameState, promise: ProfessionalPromise): void {
  assert.notEqual(promise.contractId, null)
  const employment = state.hollywood!.employment.find((row) => row.contractId === promise.contractId)
  assert.ok(employment)
  assert.equal(employment.studioId, promise.issuerStudioId)
  assert.equal(employment.terms.talentId, promise.beneficiaryPersonId)
  assert.equal(employment.terms.startWeek, promise.windowStartWeek)
  assert.ok(state.talentMarket.receipts.some((r) => r.kind === 'settled' && r.week === employment.terms.startWeek
    && r.talentId === promise.beneficiaryPersonId && r.studioId === promise.issuerStudioId))
}

function sign(state: GameState, role: 'actor' | 'writer' | 'director' | 'craft', termWeeks = 208) {
  for (let steps = 0; steps < 60; steps++) {
    const person = hiringMarketIds(state, state.market.tick).map((id) => state.talent.find((p) => p.id === id)).find((p) => p?.role === role)
    if (person !== undefined) return { state: applyActions(state, [{ kind: 'signContract', talentId: person.id, termWeeks }]), id: person.id }
    state = tick(state)
  }
  throw new Error('UNEXECUTED prerequisite: no actual hireable ' + role + ' within60 weeks')
}
const p2 = (seatClass: SeatClass) => ({ family: 'LEAD_OR_SIGNIFICANT_ROLE_COUNT' as const,
  predicate: { kind: 'castRoleCount' as const, count: 1, seatClass } })
const p1 = (count: number) => ({ family: 'APPEARANCE_COUNT' as const, predicate: { count } })

function readySoundstage(state: GameState): GameState {
  const stage = 'facility-soundstage-07'
  assert.ok(state.operations.facilities.some((f) => f.id === stage && f.capability === 'soundstage'))
  const mounted = state.sets.find((s) => s.mountedOn === stage && s.status !== 'retired')
  if (mounted !== undefined) state = applyActions(state, [{ kind: 'strikeSet', setId: mounted.id }])
  state = applyActions(state, [{ kind: 'commissionSet', commission: { blueprintId: 'set-grand-ballroom', stageFacilityId: stage } }])
  return advanceTo(state, state.market.tick + TUNING.SET_BUILD_WEEKS_BAND_HIGH)
}
/** Real submitProposal + attachPromise of a TAGGED seat-class P2, bound by the real freeze at `start`. */
function realPlayerPromise(seatClass: SeatClass) {
  let state = fund(p13aGeneratedStudio())
  const target = sign(state, 'actor', 52); state = target.state
  for (const role of ['writer', 'director', 'actor', 'actor', 'craft'] as const) state = sign(state, role).state
  const contract = state.contracts.find((c) => c.talentId === target.id)
  assert.ok(contract)
  const start = contract.endWeekExclusive
  state = readySoundstage(state)
  assert.ok(state.market.tick <= start - 7)
  state = advanceTo(state, start - 7)
  state = submitProposal(state, { talentId: target.id, issuerStudioId: player(state), termWeeks: 52, premiumTier: 1.25 })
  state = attachPromise(state, target.id, player(state), { ...p2(seatClass), windowStartWeek: start, dueWeekExclusive: start + 40 })
  const draft = ownPromise(state, target.id)
  assert.equal(draft.version, 4)
  assert.equal(draft.contractId, null)
  assert.deepEqual(draft.predicate, p2(seatClass).predicate)
  assert.equal(draft.feasibilityReceipt.rulesVersion, 4)
  assert.equal(draft.feasibilityReceipt.classification, 'REASONABLY_ACHIEVABLE')
  const submittedWeek = state.market.tick
  state = advanceTo(state, start)
  const bound = root(state, draft.promiseId)
  binding(state, bound)
  assert.equal(bound.feasibilityReceipt.rulesVersion, 4)
  assert.equal(bound.feasibilityReceipt.week, start)
  assert.equal(bound.outcome, null)
  assert.deepEqual(currentProposals(state, target.id), [])
  return { state, promiseId: draft.promiseId, talentId: target.id, submittedWeek, start,
    recipe: { builder: 'tests/p14b4-cast-class-outcomes.test.ts-shaped route (copied)', seatClass,
      generatedCampaignOnly: true, economicInput: 'explicit fund helper cash delta with matching ledger',
      actions: ['real signContract x6', 'real strikeSet/commissionSet set-grand-ballroom', `real submitProposal at ${submittedWeek}`,
        `real attachPromise LEAD_OR_SIGNIFICANT_ROLE_COUNT castRoleCount count1 ${seatClass} window [${start},${start + 40})`,
        `real market freeze at ${start} binds the promise to the winning contract`] } }
}
function playerPayload(state: GameState, promiseId: string, slot: CastSlot, concept: GameState['concepts'][number]): Extract<Action, { kind: 'greenlight' }>['production'] {
  const promised = root(state, promiseId)
  const available = state.contracts.filter((c) => c.startWeek <= state.market.tick && state.market.tick < c.endWeekExclusive)
    .map((c) => state.talent.find((p) => p.id === c.talentId)!)
  const actorIds = available.filter((p) => p.role === 'actor' && p.id !== promised.beneficiaryPersonId).map((p) => p.id)
  assert.ok(actorIds.length >= 2, 'fixture prerequisite: two lawful complementary actors missing')
  const cast: Record<CastSlot, string> = { lead: '', antagonist: '', support: '' }
  cast[slot] = promised.beneficiaryPersonId
  for (const other of SLOTS.filter((candidate) => candidate !== slot)) cast[other] = actorIds.shift()!
  const writer = available.find((p) => p.role === 'writer')
  const director = available.find((p) => p.role === 'director')
  const craft = available.find((p) => p.role === 'craft')
  assert.ok(writer && director && craft, 'fixture prerequisite: real complementary crew missing')
  assert.equal(new Set([writer.id, director.id, craft.id, ...Object.values(cast)]).size, 6)
  return { conceptId: concept.id, writerId: writer.id, directorId: director.id, cast, craftIds: [craft.id],
    shape: { opening: 'slowSetup', midpoint: 'revelation', ending: 'bittersweet' },
    promise: { genre: concept.genre, intendedSegments: ['adult'], ranges: {
      intimacy: [-0.5, 0.5], tonalWeight: [-0.5, 0.5], kineticEnergy: [-0.5, 0.5] } },
    budget: { negative: concept.baseNegativeCost, marketing: 0 } }
}
function unusedConcept(state: GameState) {
  const used = new Set([...state.studio.activeProductions.map((p) => p.conceptId),
    ...state.studio.releasedFilms.map((p) => p.conceptId), ...state.scriptDevelopment.projects.map((p) => p.conceptId)])
  const concept = state.concepts.find((c) => !used.has(c.id))
  assert.ok(concept, 'fixture prerequisite: no distinct unused existing stock concept')
  return concept
}
/** Real greenlight with the promised person in `slot`, ticked to shooting entry (remainingTicks 5), take not yet scheduled. */
function playerToFive(state: GameState, promiseId: string, slot: CastSlot) {
  binding(state, root(state, promiseId))
  const payload = playerPayload(state, promiseId, slot, unusedConcept(state))
  const greenlitWeek = state.market.tick
  state = applyActions(state, [{ kind: 'greenlight', production: payload }])
  const productionId = state.studio.activeProductions.at(-1)!.id
  assert.equal(state.firstTakes.some((take) => take.productionId === productionId), false)
  state = tick(tick(tick(state)))
  const rehearsal = state.operations.workflows.find((w) => w.productionId === productionId)
  assert.ok(rehearsal)
  assert.equal(rehearsal.phase, 'rehearsal')
  state = applyActions(state, [{ kind: 'setProductionSetupRecipe', productionId,
    recipeId: 'ballroom-reveal-lighting-01', expectedPlanRevision: rehearsal.planRevision }])
  for (let steps = 0; state.studio.activeProductions.find((p) => p.id === productionId)!.remainingTicks !== 5; steps++) {
    if (steps >= 40) throw new Error('UNEXECUTED prerequisite: player never reached shooting entry within40 actual ticks')
    state = tick(state)
  }
  const film = state.studio.activeProductions.find((p) => p.id === productionId)!
  assert.equal(film.cast[slot], root(state, promiseId).beneficiaryPersonId)
  assert.deepEqual(state.firstTakes.filter((take) => take.productionId === productionId), [])
  assert.equal(state.operations.workflows.find((w) => w.productionId === productionId)!.shootingTask?.status, 'unassigned')
  assert.equal(root(state, promiseId).outcome, null)
  return { state, productionId, film, greenlitWeek,
    recipe: { builder: 'tests/p14b4-cast-class-outcomes.test.ts-shaped route (copied)', slot,
      actions: [`real greenlight at ${greenlitWeek} (${payload.conceptId}) with the promised person as ${slot}`,
        'three real ticks to rehearsal', 'real setProductionSetupRecipe ballroom-reveal-lighting-01',
        `real ticks to remainingTicks 5 at ${state.market.tick}; shooting task unassigned; no take`] } }
}
/** Real assign + schedule + ONE tick: films the first take for real (not discarded). */
function filmOneTake(state: GameState, productionId: string, directorId: string) {
  const before = state.market.tick
  state = applyActions(state, [{ kind: 'assignShootingDirector', productionId, directorId },
    { kind: 'scheduleShootingTake', productionId }])
  state = tick(state)
  const rows = state.firstTakes.filter((take) => take.productionId === productionId)
  assert.equal(rows.length, 1)
  assert.equal(rows[0]!.week, before + 1)
  return { state, take: rows[0]! }
}
