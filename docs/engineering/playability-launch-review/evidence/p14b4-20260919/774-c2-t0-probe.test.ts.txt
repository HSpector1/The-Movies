const SEED = 'p14c2-corpus-01'

// D1, record 773 §3: windows [start, hard).
const WINDOWS: Record<string, { start: number; hard: number }> = {
  actor: { start: 60, hard: 70 },
  director: { start: 65, hard: 75 },
  writer: { start: 65, hard: 75 },
  craft: { start: 62, hard: 72 },
}

function isIdle(state: GameState, id: string, week: number): boolean {
  if (rivalEmployment(state, id, week) !== null) return false
  if (activeContract(state, id, week) !== undefined) return false
  if (busyTalentIds(state).has(id)) return false
  if (state.contracts.some((c) => c.talentId === id && c.startWeek < week && c.endWeekExclusive > week - 104)) return false
  return true
}
function anchorWeekOf(state: GameState, id: string): number | undefined {
  const row = state.talentProvenance.rows.find((r) => r.personId === id)
  return row === undefined ? undefined : anchorOf(row).week
}

function surveyWindows(state: GameState, week: number) {
  const out: Record<string, { atOrPastHard: unknown[]; idleInWindow: unknown[]; rivalInWindow: unknown[] }> = {}
  for (const role of Object.keys(WINDOWS)) out[role] = { atOrPastHard: [], idleInWindow: [], rivalInWindow: [] }
  for (const t of state.talent) {
    const w = WINDOWS[t.role]
    if (w === undefined) continue
    const bucket = out[t.role]!
    if (t.age >= w.hard) bucket.atOrPastHard.push({ id: t.id, age: t.age })
    else if (t.age >= w.start) {
      const anchorWeek = anchorWeekOf(state, t.id)
      const rival = rivalEmployment(state, t.id, week)
      if (rival !== null) bucket.rivalInWindow.push({ id: t.id, age: t.age, endWeekExclusive: rival.terms.endWeekExclusive })
      else if (anchorWeek !== undefined && anchorWeek <= week - 104 && isIdle(state, t.id, week)) {
        bucket.idleInWindow.push({ id: t.id, age: t.age, anchorWeek })
      }
    }
  }
  return out
}

function saveFacts(state: GameState, label: string) {
  try {
    const json = exportSave(makeSave(state))
    const saveVersion = validateSaveV33(JSON.parse(json)).saveVersion
    return { label, ok: true, saveVersion, bytes: Buffer.byteLength(json, 'utf8') }
  } catch (error) { return { label, ok: false, error: (error as Error).message.slice(0, 200) } }
}

function sign(state: GameState, role: 'actor' | 'writer' | 'director' | 'craft', termWeeks = 208) {
  for (let i = 0; i < 60; i++) {
    const person = hiringMarketIds(state, state.market.tick).map((id) => state.talent.find((p) => p.id === id)).find((p) => p?.role === role)
    if (person !== undefined) return { state: applyActions(state, [{ kind: 'signContract', talentId: person.id, termWeeks }] as never), id: person.id }
    state = tick(state)
  }
  throw new Error(`probe: no signable ${role}`)
}

it('MEASURES which of the C.2a corpus axes are reachable', () => {

    const windowsAt260 = Object.fromEntries(roles.map((role) => [role, {
      age: ages[role], hard: WINDOWS[role]!.hard, atOrPastHard: (ages[role] as number) >= WINDOWS[role]!.hard,
    }]))
    x1 = { reachable: true, week: 260, ages, windowsAt260, saveFacts: saveFacts(state, 'x1') }
    rows.push({ axis: 'X1-hard-boundary', ...x1 as object })

    // Reuse this SAME ticked world (natural genesis + rival population aged 5
    // years) to survey X2 (idle in-window free agents) and part of X3 (rival
    // employment landing in-window), at zero extra build cost.
    const survey260 = surveyWindows(state, 260)
    rows.push({ axis: 'X2-and-X3rival-survey-at-260', survey: survey260 })
  } catch (error) { x1 = { reachable: false, error: (error as Error).message.slice(0, 300) }; rows.push({ axis: 'X1-hard-boundary', ...x1 as object }) }

  // ── X2 supplemental — director/writer idle-in-window found 0 at week 260 on
  // the primary seed. Try more seeds and later checkpoints before giving up.
  let x2supplemental: unknown
  try {

    for (const trySeed of [SEED + '-d1', SEED + '-d2', SEED + '-d3', SEED + '-d4', SEED + '-d5']) {
      let state = p13aGeneratedStudio(trySeed)
      for (const week of [260, 520, 780, 1040]) {
        state = advanceTo(state, week)
        const survey = surveyWindows(state, week)
        for (const role of ['director', 'writer']) {
          const hits = (survey[role]!.idleInWindow as { id: string; age: number }[])
          if (hits.length > 0) found[role]!.push({ seed: trySeed, week, hits })
        }
        for (const role of ['actor', 'director', 'writer', 'craft']) {
          const hits = (survey[role]!.rivalInWindow as { id: string; age: number }[])
          if (hits.length > 0) rivalFound[role]!.push({ seed: trySeed, week, hits })
        }
      }
    }
    x2supplemental = { found, rivalFound }
  } catch (error) { x2supplemental = { error: (error as Error).message.slice(0, 300) } }
  rows.push({ axis: 'X2-supplemental-director-writer-search', ...x2supplemental as object })

  // ── X2 candidate world for the minter: full 4-profession survey at the
  // chosen seed/week combo before committing to it. ─────────────────────────
  let x2candidate: unknown
  try {
    const state = advanceTo(p13aGeneratedStudio(SEED + '-d1'), 780)
    x2candidate = { seed: SEED + '-d1', week: 780, survey: surveyWindows(state, 780), saveFacts: saveFacts(state, 'x2candidate') }
  } catch (error) { x2candidate = { error: (error as Error).message.slice(0, 300) } }
  rows.push({ axis: 'X2-candidate-world-d1-780', ...x2candidate as object })

  // ── X7 — scientist role refused by createTalent (structural: no window). ─
  let x7createAttempt: unknown
  try {
    const state = fund(p13aGeneratedStudio(SEED))
    applyActions(state, [{
      kind: 'createTalent', talent: {
        name: 'Probe Scientist', role: 'scientist', age: 60,
        actual: { warmth: 0, gravity: 0, physicality: 0 },
        potentialTier: 'Steady', workEthic: 50,
      },
    }] as never)
    x7createAttempt = { refused: false }
  } catch (error) { x7createAttempt = { refused: true, error: (error as Error).message.slice(0, 200) } }
  let x7oldest: unknown
  try {
    let state = p13aResearchReady()
    const scientist = state.talent.find((t) => t.role === 'scientist')!

  let x3x4x6: unknown
  try {

        kind: 'createTalent', talent: {

    const aNextBirthday = nextBirthdayWeek(aRow, a.age)
    const bCase = caseForTalent(state, b.id, state.market.tick)
