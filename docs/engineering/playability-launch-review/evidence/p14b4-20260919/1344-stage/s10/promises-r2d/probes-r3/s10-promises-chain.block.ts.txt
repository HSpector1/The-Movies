const s10IdNumber = (id: string): number => { const m = /^promise-(\d+)$/.exec(id); return m === null ? Number.POSITIVE_INFINITY : Number(m[1]) }
/** The rule order (1344-D7, kept by 1344-D8): r01 first, then the other rivals (1344-F5 A1); within each, cutoff week Wc,
 * then promise number. */
const s10RuleOrder = (a: S10Ordered, b: S10Ordered): number =>
  (a.issuerStudioId === ISSUER ? 0 : 1) - (b.issuerStudioId === ISSUER ? 0 : 1) || a.wc - b.wc
  || s10IdNumber(a.promiseId) - s10IdNumber(b.promiseId) || (a.promiseId < b.promiseId ? -1 : a.promiseId > b.promiseId ? 1 : 0)
/** One search's result: every candidate in rule order and the first that qualifies; ERROR when the search threw (the first
 * qualifying candidate is then undecided), NONE when none qualifies. `log` is the short form the log line prints. */
function s10Pick<T extends S10Ordered>(found: T[], errors: S10SearchError[]) {
  const candidates = [...found].sort(s10RuleOrder)
  const chosen = candidates.find((c) => c.qualifies) ?? null
  const witness = errors.length > 0 ? 'ERROR' : chosen === null ? 'NONE' : chosen
  const tag = (c: T | undefined) => c === undefined ? null : { promiseId: c.promiseId, issuerStudioId: c.issuerStudioId, wc: c.wc, qualifies: c.qualifies, checks: c.checks }
  return { witness, errors, candidates,
    log: { witness: typeof witness === 'string' ? witness : tag(witness), candidates: candidates.length, first: tag(candidates[0]), errors: errors.length } }
}
