    const dispatch = s10Dispatch(final)
    const C1old: S10Gate = { pass: anchorBase && lines.l37 && lines.l40_41 && dispatch.l142 && dispatch.l143 && dispatch.l148 && dispatch.l149
        && dispatch.l150 && dispatch.l151 && dispatch.l152,
      detail: { premise, tick: final.market.tick, retirement: record ?? null, p148: s10P148(final), productions: s10R01(final).productions.map((p) => p.id),
        rivalLines: lines, dispatch, anchor } }
    // C2 + C3: every state up to and including the one entering Ws is equal in both trees, and the first different state is the
    // one the shelving week produced, post-tick Ws + 1. The differing top-level keys locate it.
    const byPost = (s0: S10Start, ws: S10Week[]) => new Map<number, { hash: string; keys: Record<string, string> }>(
      [[s0.post, s0] as const, ...ws.map((w) => [w.post, w] as const)])
    const oldByPost = byPost(start, weeks), headByPost = byPost(head.start, head.weeks)
    let firstDiffPost: number | null = null
    let firstDiffKeys: string[] = []
    for (let post = S10_FROM; post <= S10_TO; post++) {
      const a = oldByPost.get(post), b = headByPost.get(post)
      if (a !== undefined && b !== undefined && a.hash === b.hash) continue
      firstDiffPost = post
      firstDiffKeys = a === undefined || b === undefined ? ['(missing week)']
        : [...new Set([...Object.keys(a.keys), ...Object.keys(b.keys)])].filter((k) => a.keys[k] !== b.keys[k]).sort()
      break
    }
    const P3: S10Gate = { pass: head.ws !== null && firstDiffPost === head.ws + 1,
      detail: { ws: head.ws, firstDiffPost, firstDiffKeys, startEqual: start.hash === head.start.hash } }
    // d4, old half: without shelving r01 stays stalled through 2695 and promise-148 stays open.
    const stallOld: S10Gate = { pass: startStalled && weeks.every(stalledWeek) && p148.outcome === null,
      detail: { startR01, broken: weeks.filter((w) => !stalledWeek(w)).map((w) => ({ processed: w.processed, r01: w.r01, receipts: w.receipts })) } }
    const gates: Record<string, S10Gate> = {
      C1_head_anchor: head.gates.C1_head_anchor, C1_old_anchor: C1old,
      P1_d1_r01_shelving_before_take: head.gates.P1_d1_r01_shelving_before_take,
      P2_d2_thirteen_rejections: head.gates.P2_d2_thirteen_rejections,
      P3_C2_C3_first_divergence_at_Ws_plus_1: P3,
      P4_d3_take_via_commission_or_retry: head.gates.P4_d3_take_via_commission_or_retry,
      P5_d4_stall_without_shelving: { pass: head.stallHead.pass && stallOld.pass, detail: { head: head.stallHead, old: stallOld } },
    }
    const predictions = { ...head.predictions, first_divergence_at_post_2613: firstDiffPost === 2613,
      first_divergence_in_hollywood_only: JSON.stringify(firstDiffKeys) === JSON.stringify(['hollywood']) }
    s10fs.writeFileSync(s10Out('promises.old.json'), JSON.stringify({ probe: 'S10-promises r2d', tree, premise, startR01, start, weeks }))
    s10fs.writeFileSync(s10Out('promises.predicates.json'), JSON.stringify({ probe: 'S10-promises r2d', gates, predictions,
      outcome: head.outcome, explanatory: head.explanatory }, null, 1))
    console.log('S10-promises gates ' + JSON.stringify(Object.fromEntries(Object.entries(gates).map(([k, v]) => [k, v.pass]))))
    console.log('S10-promises predictions ' + JSON.stringify(predictions))
    expect(Object.entries(gates).filter(([, v]) => !v.pass).map(([k]) => k)).toEqual([])
  }, 900_000)
