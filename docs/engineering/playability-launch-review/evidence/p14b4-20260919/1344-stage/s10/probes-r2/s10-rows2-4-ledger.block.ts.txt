function s10Out(name: string): string {
  const dir = process.env.S10_OUT_DIR ?? S10_OUT_ROOT
  const file = s10path.resolve(dir, name)
  if (!s10path.isAbsolute(dir) || !file.startsWith(S10_OUT_ROOT + '/')) throw new Error(`S10: output path outside ${S10_OUT_ROOT}: ${file}`)
  s10fs.mkdirSync(s10path.dirname(file), { recursive: true })
  return file
}

type S10Gate = { pass: boolean; detail: unknown }
type S10Row = ReturnType<typeof runLedger>['rows'][number]
function s10StateHash(state: GameState): string {
  const h = state.hollywood
  const stripped = h === null ? state : { ...state, hollywood: { ...h, businesses: h.businesses.map((b) => {
    const { screenplayShelving: _s, ...rest } = b as typeof b & { screenplayShelving?: unknown }
    return rest
  }) } }
  return sha(s10Stable(stripped))
}
/** The leaf's law lines, verbatim in substance: merged :543, :545, :546-551, :552, :555-557, :558 and :565. */
function s10LawLines(seed: string, rows: S10Row[], state: GameState, rel: Relationships) {
  const unexpected = rows.flatMap((r) => r.newSentences.map((s) => ({ eventId: r.eventId, s })))
    .filter(({ eventId, s }) => !(seed === KNOWN_EXTENSION_ROW.seed && eventId === KNOWN_EXTENSION_ROW.eventId && s === KNOWN_EXTENSION_ROW.sentence))
  return {
    unexpected,
    l543_everyRowAtChurn: rows.every((r) => r.churn),
    l545_rosters208Empty: rows.filter((x) => x.week === 208).every((r) => r.survivors.every((s) => s.roster.length === 0)),
    l546_551_rosters416: rows.filter((x) => x.week === 416).every((r) => r.survivors.every((s) =>
      s.roster.every((p) => /-supply-265-\d$/.test(p)) && s.sharedTakeCounterparts.length === 0)),
    l552_noExposedRow: rows.filter((r) => r.exposed).length === 0,
    l557_noUnexpectedSentence: unexpected.length === 0,
    l558_everyBandOne: rows.every((r) => r.survivors.every((s) => s.band === 1)),
    l565_edgeCaps: edges(state).every((e) => e.recent.length <= rel.RELATIONSHIP_RECENT_CAP && e.sharedSuccesses + e.sharedFailures <= e.sharedProductions),
  }
}
function s10FirstDiff(a: readonly { week: number }[], b: readonly { week: number }[]) {
  const n = Math.max(a.length, b.length)
  for (let i = 0; i < n; i++) {
    if (s10Stable(a[i] ?? null) !== s10Stable(b[i] ?? null)) {
      return { index: i, week: Math.min(...[a[i], b[i]].filter((x) => x !== undefined).map((x) => x!.week)), old: a[i] ?? null, head: b[i] ?? null }
    }
  }
  return null
}
