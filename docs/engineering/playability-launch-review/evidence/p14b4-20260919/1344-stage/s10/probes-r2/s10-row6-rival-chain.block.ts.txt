const S10_OUT_ROOT = '/Users/zacheryspector/studio-scratch/1344-sweep/s10/out'
function s10Out(name: string): string {
  const dir = process.env.S10_OUT_DIR ?? S10_OUT_ROOT
  const file = s10path.resolve(dir, name)
  if (!s10path.isAbsolute(dir) || !file.startsWith(S10_OUT_ROOT + '/')) throw new Error(`S10: output path outside ${S10_OUT_ROOT}: ${file}`)
  s10fs.mkdirSync(s10path.dirname(file), { recursive: true })
  return file
}
