  r3fs.mkdirSync(R3_OUT_ROOT, { recursive: true })

describe('S10 r3 row 6 re-witness (scratch only)', () => {

    const FIXED = { studioId: 'studio-aca408ec-r01', firstTake: 'studio-aca408ec-r01:film:11', firstTakeWeek: 213, firstReleaseWeek: 217 } as const
    const START = 196
    const LAST = 237
    const json = (x: unknown) => JSON.stringify(x)
    let state = lifted('rival-current-p1-and-p2') // fixture() asserts both V30_PINS sha256 pins and tick 196
    const G1_start_196 = state.market.tick === START
    // R and F per post-tick, by rivalWorld's own expressions (:387-388), for the 41 ticks its guard allows (:382-383).
    const weeks: { post: number; takes: R3Take[]; films: string[] }[] = []
    for (let guard = 0; guard <= 40; guard++) {
      const before = state
      state = tick(state)
      weeks.push({
        post: state.market.tick,
        takes: state.firstTakes.slice(before.firstTakes.length).map((t) => ({ week: t.week, studioId: t.studioId,
          productionId: t.productionId, directorId: t.directorId, cast: { ...t.cast } })),
        films: state.hollywood!.films.slice(before.hollywood!.films.length).map((f) => f.filmId),
      })

    const G5_first_take_213 = weeks.every((w) => w.post >= FIXED.firstTakeWeek || w.takes.length === 0)

      && json(films) === json(recorded.predicates.R3_d3_film11_208_213_217.detail.films)
